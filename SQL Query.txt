CREATE TABLE CONTACT_GROUPS (
    GROUP_ID NUMBER NOT NULL,
    GROUP_NAME VARCHAR2(30) NOT NULL,
    DESCRIPTION VARCHAR2(100),
    CONSTRAINT PK_CONTACT_GROUPS PRIMARY KEY (GROUP_ID)
);

CREATE TABLE CONTACTS (
    CONTACT_ID NUMBER NOT NULL,
    NAME VARCHAR2(50) NOT NULL,
    EMAIL VARCHAR2(50),
    ADDRESS VARCHAR2(100),
    GROUP_ID NUMBER,
    CONSTRAINT PK_CONTACTS PRIMARY KEY (CONTACT_ID),
    CONSTRAINT FK_CONTACTS_GROUP FOREIGN KEY (GROUP_ID) REFERENCES CONTACT_GROUPS(GROUP_ID)
);

CREATE TABLE CONTACT_PHONES (
    PHONE_ID NUMBER NOT NULL,
    CONTACT_ID NUMBER NOT NULL,
    PHONE_NUMBER VARCHAR2(15) NOT NULL,
    PHONE_TYPE VARCHAR2(20),
    CONSTRAINT PK_CONTACT_PHONES PRIMARY KEY (PHONE_ID),
    CONSTRAINT FK_PHONES_CONTACT FOREIGN KEY (CONTACT_ID) REFERENCES CONTACTS(CONTACT_ID)
);
